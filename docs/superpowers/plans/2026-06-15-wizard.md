# Interactive Wizard Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make bare `anki-enrich` in a terminal launch a friendly, apply-first interactive wizard, while every existing flag keeps working unchanged.

**Architecture:** A new `ankienrich/wizard.py` holds two pure, testable helpers (`should_run_wizard`, `_namespace_from_answers`) plus a thin `run()` shell built on `questionary`. `cli.main()` gains one branch at the top: when the trigger fires it gets its `Namespace` from the wizard instead of argparse, then falls through to the unchanged execution code.

**Tech Stack:** Python ≥3.11, `argparse`, `questionary` (new optional dep), `pytest`.

```python
# ankienrich/wizard.py
"""Interactive apply-first wizard for `anki-enrich`.

Launches on bare invocation (or sole `--dry-run`) in a TTY; every other
invocation bypasses it and runs the flag-based path unchanged.
"""


def should_run_wizard(argv: list[str], isatty: bool) -> bool:
    """True when argv is empty or exactly ['--dry-run'], and we're on a TTY."""
    return isatty and argv in ([], ["--dry-run"])
```

```python
import argparse

_FIELDS = ("pitch", "sentences", "defs", "examples", "cloze")


def _namespace_from_answers(action: str, overwrite_fields: set[str],
                            dry_run: bool) -> argparse.Namespace:
    """Build the same Namespace argparse would yield for the equivalent flags."""
    return argparse.Namespace(
        command=action,
        apply=not dry_run,
        overwrite=False,
        **{f"overwrite_{f}": (f in overwrite_fields) for f in _FIELDS},
    )
```

- [ ] **Step 1: Implement `run()`**

Add to `ankienrich/wizard.py`:

```python
import sys

_ACTIONS = [
    ("Enrich cards", "run"),
    ("Find & merge duplicates", "dedupe"),
    ("Apply card template", "template"),
    ("Quit", "quit"),
]


def _echo_command(ns: argparse.Namespace) -> None:
    """Print the equivalent flag invocation, so the flags are learnable."""
    parts = ["anki-enrich"]
    if ns.command != "run":
        parts.append(ns.command)
    if ns.apply:
        parts.append("--apply")
    for f in _FIELDS:
        if getattr(ns, f"overwrite_{f}"):
            parts.append(f"--overwrite-{f}")
    print(f"\n  → {' '.join(parts)}\n")


def run(dry_run: bool) -> argparse.Namespace | None:
    """Interactively build a Namespace. Returns None to quit/cancel."""
    try:
        import questionary
    except ImportError:
        print("  (install the `wizard` extra for the interactive menu; "
              "running a dry-run preview)")
        return _namespace_from_answers("run", set(), dry_run=True)

    action = questionary.select(
        "What do you want to do?",
        choices=[questionary.Choice(label, value) for label, value in _ACTIONS],
    ).ask()
    if action in (None, "quit"):
        return None

    fields: set[str] = set()
    if action == "run":
        picked = questionary.checkbox(
            "Re-assert any fields? (space toggles, enter confirms)",
            choices=list(_FIELDS),
        ).ask()
        if picked is None:  # Ctrl-C
            return None
        fields = set(picked)

    if not dry_run:
        ok = questionary.confirm(
            "Writes to your collection and pushes to AnkiWeb. Continue?",
            default=True,
        ).ask()
        if not ok:
            return None

    ns = _namespace_from_answers(action, fields, dry_run)
    _echo_command(ns)
    return ns
```

```python
def _apply_run_namespace():

def test_wizard_drives_main_when_triggered(monkeypatch):

        return {"scanned": 0, "backup": None, "steps": {
            "audio": {"skipped": "x"}, "pitch": {"filled": 0, "not_found": 0, "skipped": 0},
            "sentences": {"filled": 0}, "enrich": {"skipped": "x"},
            "examples": {"skipped": "x"}, "cloze": {"filled": 0, "skipped": 0}}}
```

```python
def main(argv=None) -> int:
    if argv is None:
        argv = sys.argv[1:]
    if wizard.should_run_wizard(argv, sys.stdin.isatty() and sys.stdout.isatty()):
        args = wizard.run(dry_run="--dry-run" in argv)
        if args is None:
            return 0
    else:
        ap = argparse.ArgumentParser(prog="anki-enrich",
                                     description="Post-sync Anki enrichment.")
        ap.add_argument("command", nargs="?", default="run",
                        choices=["run", "dedupe", "template"])
        ap.add_argument("--apply", action="store_true", help="write + push (else dry run)")
        ap.add_argument("--overwrite", action="store_true",
                        help="re-assert pitch, sentences, defs, examples AND cloze on every card")
        ap.add_argument("--overwrite-pitch", action="store_true")
        ap.add_argument("--overwrite-sentences", action="store_true")
        ap.add_argument("--overwrite-defs", action="store_true")
        ap.add_argument("--overwrite-examples", action="store_true")
        ap.add_argument("--overwrite-cloze", action="store_true")
        ap.add_argument("--dry-run", action="store_true",
                        help="preview only (the default); bare `--dry-run` opens the wizard")
        args = ap.parse_args(argv)

    col = collection.open_working()
```

## Notes for the implementer

- **`sys` is already imported** in `cli.py` (used by the ANSI helpers); no new
  import needed there beyond `wizard`.
- **`argparse` import stays** in `cli.py` — it's still used in the `else` branch.
- The wizard never re-parses argv through argparse; it constructs the
  `Namespace` directly, so the two paths must stay attribute-compatible. The
  `_namespace_from_answers` tests in Task 2 lock that contract.
- Flag mode is unchanged: default is dry-run, `--apply` writes. `--dry-run` is a
  recognized no-op there (preview is already the default).
